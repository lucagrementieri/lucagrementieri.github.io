{{- /* Aligns the notes of each paragraph with the start of the paragraph on wide screens: a copy of the notes is put at the beginning of the paragraph and shown only on wide screens, while the notes in their original position are shown only on narrow screens, where they appear inline next to the text they refer to. Hidden copies use display: none, so they are skipped by screen readers. Context: the rendered content of a page. Returns the transformed content. */}} {{- $content := string . }} {{- range $paragraph := findRE `(?s)

.*?

` $content }} {{- with findRE `(?s).*?` $paragraph }} {{- $copies := delimit . "" | replaceRE `` "" }} {{- $inline := $paragraph | replaceRE `` "" }} {{- $content = replace $content $paragraph (printf "

%s%s" $copies (strings.TrimPrefix "

" $inline)) }} {{- end }} {{- end }} {{- $content = $content | replaceRE `` "" | safeHTML }} {{- return $content }}
